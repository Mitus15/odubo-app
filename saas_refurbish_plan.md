 Executive Summary                                                                                                                                                                                                                                          
                                                                                                                                                                                                                                                            
 Build "The Hub" - an enterprise-grade, mobile-first business operating system at /src/app/admin-v2. This is a parallel admin system that preserves the existing /admin while creating a comprehensive command center for a fashion/entertainment company.  
                                                                                                                                                                                                                                                            
 Build Strategy: Depth-first (complete Phase 1 fully before proceeding)                                                                                                                                                                                     
 Routing: Path-based (/admin-v2) initially, subdomain (admin.odubo.studio) later                                                                                                                                                                            
 Design: Dark luxury aesthetic (warm brown, glass morphism - current brand vibe)                                                                                                                                                                            
 Primary Usage: Mobile (team is in the field at events/shoots)

 -- Role definitions                                                                                                                                                                                                                                        
 CREATE TABLE IF NOT EXISTS role_definitions (                                                                                                                                                                                                              
   slug TEXT PRIMARY KEY,                                                                                                                                                                                                                                   
   name TEXT NOT NULL,                                                                                                                                                                                                                                      
   description TEXT,                                                                                                                                                                                                                                        
   default_modules TEXT, -- JSON array                                                                                                                                                                                                                      
   default_permissions TEXT, -- JSON object                                                                                                                                                                                                                 
   is_system INTEGER DEFAULT 0,                                                                                                                                                                                                                             
   created_at TEXT DEFAULT CURRENT_TIMESTAMP                                                                                                                                                                                                                
 );                                                                                                                                                                                                                                                         
                                                                                                                                                                                                                                                            
 -- User role assignments                                                                                                                                                                                                                                   
 CREATE TABLE IF NOT EXISTS user_roles (                                                                                                                                                                                                                    
   id INTEGER PRIMARY KEY AUTOINCREMENT,                                                                                                                                                                                                                    
   user_id TEXT NOT NULL,                                                                                                                                                                                                                                   
   role_slug TEXT NOT NULL,                                                                                                                                                                                                                                 
   module_access TEXT, -- JSON array (override)                                                                                                                                                                                                             
   permissions TEXT, -- JSON object (override)                                                                                                                                                                                                              
   assigned_by TEXT,                                                                                                                                                                                                                                        
   created_at TEXT DEFAULT CURRENT_TIMESTAMP,                                                                                                                                                                                                               
   UNIQUE(user_id, role_slug),                                                                                                                                                                                                                              
   FOREIGN KEY (user_id) REFERENCES users(id)                                                                                                                                                                                                               
 );
